\chapter{Analysedossier TPPBAMT}%
\label{ch:bijlage-views}

Deze bijlage bundelt uittreksels uit het analysedossier dat tijdens de reverse engineering van de
CA~Gen-module \texttt{TPPBAMT} (Hoofdstuk~\ref{ch:reverse-engineering}) werd opgesteld. Omwille van de omvang
en het bedrijfsgevoelige karakter van het volledige dossier worden hier enkel representatieve fragmenten
opgenomen; het volledige dossier is intern beschikbaar bij ArcelorMittal.

\section{View-inventaris (uittreksel)}
\label{sec:bijlage-view-inventaris}

\autoref{tab:bijlage-view-inventaris} toont een uittreksel van de view-inventaris voor het hoofd-Action
Block van \texttt{TPPBAMT}, opgesteld volgens de template uit \autoref{tab:view-inventaris}.

\begin{table}[h]
\centering
\caption[Uittreksel view-inventaris TPPBAMT]{Uittreksel uit de view-inventaris van \texttt{TPPBAMT}}
\label{tab:bijlage-view-inventaris}
\small
\begin{tabular}{|l|l|l|p{0.30\linewidth}|}
\hline
\textbf{Viewnaam} & \textbf{Categorie} & \textbf{Onderliggende structuur} & \textbf{Opmerkingen} \\
\hline
\texttt{IMP\_BAF} & Import & Work View & Werkcontext (oven, horizon, proces). \\
\hline
\texttt{EXP\_BAF} & Export & Work View & Exitstate, aantal segmenten. \\
\hline
\texttt{LCL\_TELLER} & Local & Work View & Tellers en stuurvariabelen. \\
\hline
\texttt{INT\_MAT\_MATERIAALEENHEID} & Entity & MAT & Buffer voor READ-acties op materialen. \\
\hline
\texttt{GRL\_MAT\_PRDO} & Group (entity) & MAT $\times$ PRDO & Lijst planbare materialen + orders. \\
\hline
\texttt{INT\_AFF\_TTI} & Entity & Werkveld-allocatie & Buffer voor sequenti\"ele werkveldtoekenning. \\
\hline
\end{tabular}
\end{table}

\label{sec:bijlage-usegraaf}

\begin{center}
    \captionof{listing}{USE-graaf van Action Block \texttt{AB\_APP\_GBP\_MAT\_BAF}}
    \label{lst:USE-graaf}
\end{center}

\begin{minted}[fontsize=\small]{text}
    
    AB_APP_GBP_MAT_BAF
    
    READ RO, MAT
    
    USE eab_oproep_tpbafgg   /* ophalen bestelgegevens
    
    USE ipo_rta_bep_routebewerking_info /* check of skp op routing voorkomt
      USE eab_oproep_tpolie   /* haal oliegegevens

    USE ab_scan_materiaal_stamboom   /* scan materiaal stamboom
    
    USE eab_oproep_ypglkro   /* bepaal gloeikarakteristieken
    
    USE eab_oproep_tpgghal   /* bepaal geschiktheid ligging
    
    USE ab_bepaal_aantal_dagen_stock /* bepaal aantal dagen in stock
    
    USE ab_bepaal_mogelijke_skinpass   /* bepaal mogelijke skinpass
    
    USE ab_bep_dlstk_beschikbaarheid_baf   /* bepaal deelstock/beschikbaarheid
    
    USE ipo_mat_lees_ly_lst_prc_eh_mat   /* bepaal laatste proces voor het gloeien
    
    USE ipo_mop_lees_mat_opm_met_positie   /* zoeken naar opmerking voor kleefkritische rollen
    
    USE ipo_mop_lees_mat_opm_met_positie   /* bepaal alle opmerkingen van het materiaal
    
    USE eab_oproep_tprton                  /* bepaal soort materiaal
      USE ipo_vre_haal_vreemd_nummer
    
    USE eab_oproep_mat071   /* ophalen laatste PRCZINK
      USE eab_oproep_mat057_lees_prcruw
      USE eab_oproep_mat071
    
    USE eab_oproep_mat105   /* bepalen max & min breedte
    
    USE eab_oproep_tpdiam   /* voor uitwendige diameter
    
    USE eab_oproep_tpinwdi   /* module voor de berekening van de inwendige diameter aan de uitgang van het proces
    
    USE eab_oproep_tpmagda   /* bepalen geforceerde parameters MAGDA
    
    USE ab_lpl_bepaal_lijnmogelijkheden
    
    USE ab_lpl_app_pp_sgm_plnmat   /* relatie naar PLNRT	
    
    USE ab_app_pp_sgm_bafmat_baf
